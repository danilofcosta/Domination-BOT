import { fileURLToPath } from "url";
import { MediaType } from "../generated/prisma/enums.js";
import { prisma } from "../lib/prisma";
import { getMedia } from "../src/handlers/Comandos/admin_bot/manager_character/add/add_charecter";
import { ChatType, type MyContext } from "../src/utils/customTypes";
import { create_caption } from "../src/utils/manege_caption/create_caption";
import { Sendmedia } from "../src/utils/sendmedia";
import * as fs from "fs";
import * as path from "path";
let cont = 0;

const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);
export const localesDir = path.join(__dirname, "a");

const filePath = path.join(localesDir, "waifus.json");
export async function Addchatmanual(ctx: MyContext) {
  const waifus = await prisma.characterWaifu.findMany({
    where: {
      mediaType: {
        in: [MediaType.IMAGE_LOCAL, MediaType.VIDEO_LOCAL],
      },
    },
    orderBy: {
      id: "asc",
    },
    include: {
      WaifuEvent: {
        include: {
          Event: true,
        },
      },
      WaifuRarity: {
        include: {
          Rarity: true,
        },
      },
    },
  });

  // array final
  const data = {
    waifu: [] as {
      id: number;
      slug: string;
      file_id: {
        file_id: string;
        type: MediaType;
      } | null;
      type: MediaType;
    }[],
  };

  for (const character_db of waifus) {
    try {
      let caption = create_caption({
        character: character_db,
        chatType: ChatType.WAIFU,
        t: ctx.t,
        noformat: false,
      });

      const slug = character_db.slug;

      const midia = await Sendmedia({
        ctx,
        chat_id: process.env.DATABASE_TELEGREM_ID!,
        caption,
        per: character_db,
      });

      console.log(character_db.id);
     
      console.log("Contador:", cont, "fulllname:", waifus.length);

      let file_id = getMedia(midia);
      file_id = file_id ?? {
        file_id: "",
        type: character_db.mediaType,
      };
      console.log("file_id:", file_id);

      // adiciona no array
      data.waifu.push({
        id: character_db.id,
        slug,
        file_id,
        type: character_db.mediaType,
      });
     
     
      if (cont % 20 === 0) {
        console.log(
          `Salvando backup... (${cont} registros processados)`,
          path.join(localesDir, "waifus.json"),
        );
        fs.writeFileSync(
          path.join(localesDir, "waifus.json"),

          JSON.stringify(data, null, 2),
          "utf-8",
        );
        await new Promise((resolve) => setTimeout(resolve, 5000));
        console.log(`Backup salvo (${cont} registros)`);
      }

      await new Promise((resolve) => setTimeout(resolve, 2000)); // espera 1segundo entre cada iteração
    } catch (error) {
      fs.writeFileSync(
        path.join(localesDir, "erro.json"),
        JSON.stringify(data, null, 2),
        "utf-8",
      );
      console.error(
        `Erro ao processar personagem ${character_db.slug}:`,
        error,
      );
    }
 cont++;

    // caminho do arquivo
  
    // salva JSON formatado
  }
    fs.writeFileSync(filePath, JSON.stringify(data, null, 2), "utf-8");
}
